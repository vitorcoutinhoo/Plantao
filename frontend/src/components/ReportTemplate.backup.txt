interface ReportRow {
  colaborador: string;
  inicio: string;
  fim: string;
  status: string;
  valor: number | null;
}

interface ReportTemplateOptions {
  rows: ReportRow[];
  periodoInicio: string;
  periodoFim: string;
  funcionarioNome: string;
  emitidoEm: string;
  totalValor: number;
  logoBase64: string;
}

const statusLabel: Record<string, string> = {
  PENDENTE: 'Aberto', APROVADO: 'Pago', RECUSADO: 'Recusado',
  '0': 'Aberto', '1': 'Pago', '2': 'Recusado',
};

const formatDate = (iso: string) => {
  if (!iso) return '—';
  const d = new Date(iso);
  return isNaN(d.getTime()) ? '—' : d.toLocaleDateString('pt-BR');
};

export function buildReportHTML(opts: ReportTemplateOptions): string {
  const { rows, periodoInicio, periodoFim, funcionarioNome, emitidoEm, totalValor, logoBase64 } = opts;

  const tbody = rows.map((r, i) => {
    const sc = r.status === 'APROVADO' || r.status === '1' ? '#15803d'
      : r.status === 'RECUSADO' || r.status === '2' ? '#b91c1c' : '#c2410c';
    const sbg = r.status === 'APROVADO' || r.status === '1' ? '#f0fdf4'
      : r.status === 'RECUSADO' || r.status === '2' ? '#fef2f2' : '#fff7ed';
    const bg = i % 2 === 0 ? '#fff' : '#f9fafb';
    return `<tr style="background:${bg}">
      <td class="nome">${r.colaborador}</td>
      <td>${formatDate(r.inicio)}</td>
      <td>${formatDate(r.fim)}</td>
      <td><span class="badge" style="background:${sbg};color:${sc}">${statusLabel[r.status] ?? r.status}</span></td>
      <td class="r">${r.valor != null ? `R$ ${r.valor.toFixed(2)}` : '—'}</td>
    </tr>`;
  }).join('');

  const year = new Date().getFullYear();
  const logoTag = logoBase64
    ? `<img src="${logoBase64}" alt="Intellisys" style="height:22px"/>`
    : `<span style="font-size:13pt;font-weight:800;color:#B5071A">Intellisys</span>`;

  return `<!DOCTYPE html><html lang="pt-BR"><head><meta charset="UTF-8"/>
<style>
  *{box-sizing:border-box;margin:0;padding:0}
  @page{size:A4;margin:0}
  html,body{width:210mm;font-family:Arial,sans-serif;font-size:9pt;color:#111;background:#fff}

  #header{
    position:fixed;top:0;left:0;right:0;
    padding:10mm 14mm 6mm 14mm;
    background:#fff;
    display:flex;flex-direction:column;gap:8px;
  }
  #footer{
    position:fixed;bottom:0;left:0;right:0;
    padding:3mm 14mm 8mm 14mm;
    background:#fff;
    text-align:center;
  }
  #footer-sep{
    display:block;
    width:60%;
    height:1px;
    background:#e5e7eb;
    margin:0 auto 5px auto;
  }
  #content{
    margin-bottom:20mm;
    padding:0 14mm;
  }

  .logo-bloco{display:flex;align-items:center;gap:14px;border:1px solid #e5e7eb;border-radius:8px;padding:9px 14px}
  .logo-sep{width:1px;height:22px;background:#e5e7eb;flex-shrink:0}
  .logo-titulo{font-size:10.5pt;font-weight:bold;color:#111}
  .logo-sub{font-size:6.5pt;color:#9ca3af;text-transform:uppercase;letter-spacing:.07em;margin-top:2px}

  .info-bloco{display:flex;justify-content:space-between;align-items:center;padding:8px 14px}
  .info-campos{display:flex;gap:32px}
  .campo-label{font-size:6.5pt;text-transform:uppercase;letter-spacing:.08em;color:#9ca3af;margin-bottom:3px}
  .campo-valor{font-size:9.5pt;font-weight:bold;color:#111827}
  .total-label{font-size:6.5pt;text-transform:uppercase;letter-spacing:.08em;color:#9ca3af;margin-bottom:3px;text-align:right}
  .total-valor{font-size:14pt;font-weight:800;color:#B5071A;text-align:right}

  .rodape-linha1{font-size:8pt;color:#374151;font-weight:600;margin-bottom:2px}
  .rodape-linha2{font-size:7pt;color:#9ca3af}

  .tabela-bloco{border:1px solid #e5e7eb;border-radius:8px;overflow:hidden}
  table{width:100%;border-collapse:collapse;font-size:9pt;table-layout:fixed}
  col.c-func{width:35%}col.c-data{width:15%}col.c-status{width:20%}col.c-valor{width:15%}
  thead tr{background:#B5071A}
  th{color:#fff;font-size:8pt;font-weight:700;text-transform:uppercase;letter-spacing:.08em;padding:11px 14px;text-align:left}
  th.r{text-align:right}
  td{padding:7px 14px;border-bottom:1px solid #f3f4f6;color:#374151;font-size:9pt}
  td.nome{font-weight:600;color:#111827}
  td.r{text-align:right;font-weight:600;color:#111827}
  .badge{display:inline-block;padding:2px 10px;border-radius:20px;font-size:7pt;font-weight:700;text-transform:uppercase}
  tfoot tr{background:#f9fafb;border-top:2px solid #e5e7eb}
  tfoot td{color:#6b7280;font-size:8pt;font-weight:600;padding:7px 14px;border:none}
  tfoot td.r{text-align:right;color:#111827;font-size:9pt;font-weight:800}

  *{-webkit-print-color-adjust:exact!important;print-color-adjust:exact!important}
</style></head><body>

<div id="header">
  <div class="logo-bloco">
    ${logoTag}
    <div class="logo-sep"></div>
    <div>
      <div class="logo-titulo">Relatório de Escala de Plantões</div>
      <div class="logo-sub">Intellisys Informática Ltda</div>
    </div>
  </div>
  <div class="info-bloco">
    <div class="info-campos">
      <div><div class="campo-label">Período</div><div class="campo-valor">${periodoInicio} → ${periodoFim}</div></div>
      <div><div class="campo-label">Funcionário</div><div class="campo-valor">${funcionarioNome}</div></div>
      <div><div class="campo-label">Plantões</div><div class="campo-valor">${rows.length}</div></div>
    </div>
    <div>
      <div class="total-label">Valor Total</div>
      <div class="total-valor">R$ ${totalValor.toFixed(2)}</div>
    </div>
  </div>
</div>

<div id="footer">
  <span id="footer-sep"></span>
  <div class="rodape-linha1">Intellisys Informática Ltda — Todos os direitos reservados © ${year}</div>
  <div class="rodape-linha2">Documento gerado automaticamente pelo sistema de gestão de plantões &nbsp;·&nbsp; Emitido em: ${emitidoEm}</div>
</div>

<div id="content">
  <div class="tabela-bloco">
    <table>
      <colgroup>
        <col class="c-func"/><col class="c-data"/><col class="c-data"/><col class="c-status"/><col class="c-valor"/>
      </colgroup>
      <thead><tr>
        <th>Funcionário</th><th>Início</th><th>Fim</th><th>Status</th><th class="r">Valor</th>
      </tr></thead>
      <tbody>${tbody}</tbody>
      <tfoot><tr>
        <td colspan="3">${rows.length} plantão(ões) no período</td>
        <td>Total Geral</td>
        <td class="r">R$ ${totalValor.toFixed(2)}</td>
      </tr></tfoot>
    </table>
  </div>
</div>

<script>
  window.onload = function() {
    var h = document.getElementById('header').offsetHeight;
    document.getElementById('content').style.marginTop = (h + 4) + 'px';
    window.print();
    window.onafterprint = function(){ window.close(); };
  };
</script>
</body></html>`;
}
